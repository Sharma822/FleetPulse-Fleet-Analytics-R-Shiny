-- 1. Table Creation
DROP TABLE EMPLOYEE_SALARY CASCADE CONSTRAINTS;

CREATE TABLE EMPLOYEE_SALARY (
    EMP_ID       NUMBER PRIMARY KEY,
    EMP_NAME     VARCHAR2(50),
    BASIC_SALARY NUMBER(10, 2)
);

-- 2. Bonus Function
CREATE OR REPLACE FUNCTION CALCULATE_BONUS (
    P_SALARY NUMBER
)
RETURN NUMBER
IS
    V_BONUS NUMBER;
BEGIN
    IF P_SALARY IS NULL OR P_SALARY <= 0 THEN
        V_BONUS := 0;
    ELSIF P_SALARY < 30000 THEN
        V_BONUS := P_SALARY * 0.05;
    ELSIF P_SALARY >= 30000 AND P_SALARY < 60000 THEN
        V_BONUS := P_SALARY * 0.10;
    ELSE
        V_BONUS := P_SALARY * 0.15;
    END IF;

    RETURN V_BONUS;
END;

-- 3. Run in SQL Query
SELECT
    EMP_ID,
    EMP_NAME,
    BASIC_SALARY,
    CALCULATE_BONUS(BASIC_SALARY) AS BONUS_AMOUNT,
    (BASIC_SALARY + CALCULATE_BONUS(BASIC_SALARY)) AS TOTAL_PAYOUT
FROM
    EMPLOYEE_SALARY;

-- 4. Run in PL/SQL Block
DECLARE
    V_EMP_SALARY   NUMBER := 45000;
    V_BONUS_AMT    NUMBER;
    V_TOTAL_PAYOUT NUMBER;
BEGIN
    V_BONUS_AMT    := CALCULATE_BONUS(V_EMP_SALARY);
    V_TOTAL_PAYOUT := V_EMP_SALARY + V_BONUS_AMT;

    DBMS_OUTPUT.PUT_LINE('Basic Salary = ' || V_EMP_SALARY);
    DBMS_OUTPUT.PUT_LINE('Bonus Amount = ' || V_BONUS_AMT);
    DBMS_OUTPUT.PUT_LINE('Total Payout = ' || V_TOTAL_PAYOUT);
END;
